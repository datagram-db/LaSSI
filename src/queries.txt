// query 9BIS: If the nome/adjective pair is related to a given concept, then this can be grouped as one single variable and represent the relatedness as a RelatedTo
nested match  FBinaryPredicate("isA", var(w), FVariable (var(y), var(a) ,var(b),var(c),var(d),var(e)) , var(score), var(props))
       extend-with "getOutgoingNodes" with variable : "b"   result : "z"   rel : "isA"
       as deep rewrite jsonpath "$0:$.dst.specification" to var(z)
       ;


nested match FUnaryPredicate (var(rel), var(arg), var(score), var(props) )
       extend-with "extractProperties" with from : "$0" field : "SPACE" result : "loc"
       as FBinaryPredicate ("locatedIn", var(arg) , var(loc), var(score), ( "removeFromProperties" ( var(props) ) with key : "SPACE" )  )
;


nested match FUnaryPredicate (var(rel), var(arg), var(score), var(props) )
       extend-with "extractProperties" with from : "$0" field : "SPACE" result : "loc",
                   "extractProperties" with from : "$0" field : "TOGETHERNESS" result : "friend"
       as FBinaryPredicate ("locatedIn", var(friend) , var(loc), var(score), ( "removeFromProperties" ( "removeFromProperties" ( var(props) ) with key : "TOGETHERNESS" ) with key : "SPACE" )  )
